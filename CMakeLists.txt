cmake_minimum_required(VERSION 3.16)
project(tgs C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_compile_options(-Wall -Wextra -Wpedantic)

enable_testing()

#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# deps/lvgl  (LVGL v9.6.x)
#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# Tell LVGL where to find our lv_conf.h (src/backends/lvgl/lv_conf.h).
# LVGL's CMake will define LV_CONF_INCLUDE_SIMPLE and add this dir to includes.
set(LV_BUILD_CONF_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/backends/lvgl"
    CACHE PATH "Directory containing lv_conf.h")
# Disable LVGL features TGS doesn't need
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "Build LVGL demos" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "Build LVGL examples" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "Enable ThorVG" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "Internal ThorVG" FORCE)
add_subdirectory(deps/lvgl)

#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# deps/libsixel  (autotools project — no CMakeLists.txt shipped)
# Compile sources directly into a static library.
#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Generate sixel.h from sixel.h.in (@PACKAGE_VERSION@, @LS_LTVERSION@)
set(PACKAGE_VERSION "1.8.7-r2")
set(LS_LTVERSION "1:6:0")
set(LIBSIXEL_GEN_DIR "${CMAKE_BINARY_DIR}/libsixel/gen")
file(MAKE_DIRECTORY ${LIBSIXEL_GEN_DIR}/include)
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/deps/libsixel/include/sixel.h.in
    ${LIBSIXEL_GEN_DIR}/include/sixel.h
    @ONLY)

set(LIBSIXEL_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/deps/libsixel/src)

set(LIBSIXEL_SOURCES
    ${LIBSIXEL_SRC_DIR}/output.c
    ${LIBSIXEL_SRC_DIR}/fromsixel.c
    ${LIBSIXEL_SRC_DIR}/tosixel.c
    ${LIBSIXEL_SRC_DIR}/quant.c
    ${LIBSIXEL_SRC_DIR}/dither.c
    ${LIBSIXEL_SRC_DIR}/frame.c
    ${LIBSIXEL_SRC_DIR}/pixelformat.c
    ${LIBSIXEL_SRC_DIR}/scale.c

    ${LIBSIXEL_SRC_DIR}/fromgif.c

    ${LIBSIXEL_SRC_DIR}/decoder.c
    ${LIBSIXEL_SRC_DIR}/writer.c
    ${LIBSIXEL_SRC_DIR}/stb_image_write.c
    ${LIBSIXEL_SRC_DIR}/status.c
    ${LIBSIXEL_SRC_DIR}/malloc_stub.c
    ${LIBSIXEL_SRC_DIR}/allocator.c
    ${LIBSIXEL_SRC_DIR}/tty.c
)

add_library(sixel STATIC ${LIBSIXEL_SOURCES})
target_include_directories(sixel PUBLIC
    ${LIBSIXEL_GEN_DIR}/include    # generated sixel.h
    ${LIBSIXEL_GEN_DIR}            # generated config.h
    ${LIBSIXEL_SRC_DIR}            # source-level headers
)
target_link_libraries(sixel PUBLIC m)

#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# TGS targets
#━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# tgs_common — protocol + backend abstraction
add_library(tgs_common STATIC
    src/common/tgs_frame.c
    src/common/tgs_backend.c
)
target_include_directories(tgs_common PUBLIC src/common)

# tgs_client — C client library
add_library(tgs_client STATIC
    src/client/tgs_client.c
)
target_include_directories(tgs_client PUBLIC src/client)
target_link_libraries(tgs_client PUBLIC tgs_common)

# tgs_lvgl_backend — LVGL + sixel rendering
add_library(tgs_lvgl_backend STATIC
    src/backends/lvgl/lvgl_backend.c
    src/backends/lvgl/lvgl_sixel.c
)
target_include_directories(tgs_lvgl_backend PUBLIC src/backends/lvgl)
target_link_libraries(tgs_lvgl_backend PUBLIC tgs_common lvgl sixel)

# tgs_compositor — compositor library
add_library(tgs_compositor STATIC
    src/compositor/parser.c
    src/compositor/window_manager.c
    src/compositor/event_engine.c
    src/compositor/output.c
)
target_include_directories(tgs_compositor PUBLIC src/compositor)
target_link_libraries(tgs_compositor PUBLIC tgs_common)

# tgs-compositor — executable
add_executable(tgs-compositor
    src/compositor/main.c
)
target_link_libraries(tgs-compositor PRIVATE
    tgs_compositor tgs_lvgl_backend util)

# simple_form — example app
add_executable(simple_form
    examples/simple_form.c
)
target_link_libraries(simple_form PRIVATE tgs_client)

# ime_app — IME engine (separate TGS app)
add_executable(ime_app
    examples/ime_app.c
)
target_link_libraries(ime_app PRIVATE tgs_client)

# test_frame
add_executable(test_frame
    tests/test_frame.c
)

# test_protocol
add_executable(test_protocol
    tests/test_protocol.c
)
target_link_libraries(test_protocol PRIVATE tgs_common)
add_test(NAME test_protocol COMMAND test_protocol)
